cmake_minimum_required(VERSION 3.16)
project(SysScope VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
add_library(syscope_core src/syscope.cpp src/sampler.cpp)
target_include_directories(syscope_core PUBLIC include)
target_link_libraries(syscope_core PUBLIC Threads::Threads)
add_executable(syscope src/main.cpp)
target_link_libraries(syscope PRIVATE syscope_core)

function(syscope_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
syscope_warnings(syscope_core)
syscope_warnings(syscope)

include(CTest)
if(BUILD_TESTING)
    add_executable(syscope_tests tests/tests.cpp)
    target_link_libraries(syscope_tests PRIVATE syscope_core)
    syscope_warnings(syscope_tests)
    add_test(NAME core COMMAND syscope_tests)
    add_test(NAME sampler COMMAND syscope_tests --sampler)
    set_tests_properties(sampler PROPERTIES TIMEOUT 15)
    add_test(NAME cli_help COMMAND syscope --help)
    add_test(NAME cli_invalid_duration COMMAND syscope log --duration -1 --output unused.csv)
    set_tests_properties(cli_invalid_duration PROPERTIES WILL_FAIL TRUE)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_test(NAME live_proc COMMAND syscope_tests --live)
        add_test(NAME live_monitor COMMAND syscope monitor --duration 2)
        add_test(NAME live_processes COMMAND syscope processes)
        add_test(NAME live_csv COMMAND ${CMAKE_COMMAND}
            -DSYSCOPE=$<TARGET_FILE:syscope> -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/smoke.csv
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/log_smoke.cmake)
    else()
        message(STATUS "Live monitoring requires Linux /proc; parser tests are portable")
    endif()
endif()
